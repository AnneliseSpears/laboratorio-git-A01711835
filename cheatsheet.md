# Git
git status: shows current status of folder (what is modified, etc.)\n
git commit: records current work
git push: uploads work to Github
git pull: retrieves saved work
git log: commit history
git add nombre: adds changes to specific file
git add .: adds all changes
git diff: compares changes to file (not yet staged)
git diff --staged: shows staged changes (ready to commit)
git restore: undoes changes and loses work
git restore --staged: undoes changes and keeps work

# Other
pwd: route of current folder
ls: everything inside that folder
cd nombre: move to a folder
cd ..: move backward a folder
cd ../..: move backward two folders
mkdir: create a new folder
